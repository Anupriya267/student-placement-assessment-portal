/* ===================================================== REAL SERVER-SIDE JAVA CODING EVALUATOR ===================================================== */ async function evaluateCoding( code, question ){ if( !code || !code.trim() ){ return { status: "Wrong", score: 0, message: "No Java code submitted." }; } /* ----------------------------------------------------- Read coding information from the Excel question. ----------------------------------------------------- */ const sampleInput = getExcelField( question, [ "Sample Input", "Sample Input Format", "Input", "Student Input Format" ] ) || ""; const expectedOutput = getExcelField( question, [ "Sample Output", "Expected Output", "Output", "Student Output Format" ] ) || ""; const requirement = getExcelField( question, [ "Coding Requirement", "Requirement", "Question" ] ) || ""; const evaluationCriteria = getExcelField( question, [ "Full Mark Criteria", "Full-Mark Logic / Answer Key", "Evaluation Criteria", "Coding Evaluation" ] ) || ""; const payload = { code: code, question: requirement, sampleInput: sampleInput, expectedOutput: expectedOutput, evaluationCriteria: evaluationCriteria }; try{ const url = GOOGLE_SCRIPT_URL + "?action=evaluateJava&data=" + encodeURIComponent( JSON.stringify( payload ) ); const response = await fetch( url, { method: "GET", cache: "no-store" } ); if( !response.ok ){ throw new Error( "Coding evaluation server error: " + response.status ); } const result = await response.json(); if( !result ){ throw new Error( "Empty evaluation result." ); } return { status: result.status || "Wrong", score: Number( result.score || 0 ), message: result.message || "", actualOutput: result.actualOutput || "", expectedOutput: result.expectedOutput || "", compileError: result.compileError || "", runtimeError: result.runtimeError || "" }; } catch(error){ console.error( "Java coding evaluation failed:", error ); /* IMPORTANT: Do not falsely give marks when the evaluator itself is unavailable. */ return { status: "Wrong", score: 0, message: "Unable to evaluate Java code on the server." }; } }
